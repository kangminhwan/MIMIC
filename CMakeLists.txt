cmake_minimum_required(VERSION 3.24)
project(MIMIC LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The starter TCP/WinHTTP transport targets Windows. Replace Netlib for other hosts.")
endif()
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Protobuf REQUIRED)
protobuf_generate_cpp(PROTO_SRCS PROTO_HDRS ProtocolBuffer/proto/mimic.proto)
add_library(mimic_protocol ${PROTO_SRCS} ${PROTO_HDRS})
target_include_directories(mimic_protocol PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(mimic_protocol PUBLIC protobuf::libprotobuf)
add_library(mimic_holdem Server/TableServer/Holdem/Table.cpp)
target_include_directories(mimic_holdem PUBLIC Server/TableServer)
target_link_libraries(mimic_holdem PUBLIC mimic_protocol bcrypt)
add_executable(Mimic.TableServer Server/TableServer/main.cpp Server/Netlib/Transport.cpp)
target_link_libraries(Mimic.TableServer PRIVATE mimic_holdem ws2_32 winhttp)
target_include_directories(Mimic.TableServer PRIVATE Server)
if(MSVC)
  target_compile_options(mimic_holdem PRIVATE /W4 /utf-8)
  target_compile_options(Mimic.TableServer PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_executable(Mimic.Holdem.Tests tests/HoldemTests.cpp)
target_link_libraries(Mimic.Holdem.Tests PRIVATE mimic_holdem)
add_test(NAME holdem COMMAND Mimic.Holdem.Tests)
